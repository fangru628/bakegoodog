        // 計算單筆折扣後金額

        function calculateCartSummary() {

            const count = cart.length;

            if (count === 0) return { rawSum: 0, discount: 0, finalTotal: 0, formulaParts: [] };


            let rawSum = 0;

            let discount = 0;

            let formulaParts = [];


            if (isFriendActive && friendConfig.mode === 'per_item') {

                const perOff = Number(friendConfig.value) || 0;

                cart.forEach(item => {

                    rawSum += item.price;

                    const discountedItemPrice = Math.max(0, item.price - perOff);

                    formulaParts.push(discountedItemPrice);

                });

                const finalTotal = formulaParts.reduce((a, b) => a + b, 0);

                discount = rawSum - finalTotal;

                return { rawSum, discount, finalTotal, formulaParts };

            } else {

                cart.forEach(item => {

                    rawSum += item.price;

                    formulaParts.push(item.price);

                });

                if (isFriendActive) {

                    if (friendConfig.mode === 'order_flat') {

                        discount = Number(friendConfig.value) || 0;

                    } else if (friendConfig.mode === 'percent') {

                        const pct = Number(friendConfig.value) || 100;

                        const afterPct = Math.round(rawSum * (pct / 100));

                        discount = Math.max(0, rawSum - afterPct);

                    }

                }

                const finalTotal = Math.max(0, rawSum - discount);

                return { rawSum, discount, finalTotal, formulaParts };

            }

        }

                    expr += `-${discount}`;

                if (cashInput !== '') {

                    expr = `應收${finalTotal} / 實收${cashInput}`;

                formulaEl.innerText = expr;

            // 3. 右下角大數字

        // 確認鍵（完成結帳、扣庫存、記錄交易、累加今日收入）

        function confirmCheckout() {

            if (cart.length === 0) {

                showNotice('尚未選取品項', '請先點選左側的甜點品項再按確認結帳喔！');

                return;

            }


            const { finalTotal, discount } = calculateCartSummary();

            let changeHtml = '';


            if (cashInput !== '') {

                const paid = parseInt(cashInput, 10);

                if (paid < finalTotal) {

                    showNotice('實收金額不足', `實收 $${paid} 小於應收金額 $${finalTotal}！`);

                    return;

                }

                const change = paid - finalTotal;

                changeHtml = `<br>實收金額：$${paid}<br><span class="text-lg underline">應找零：$${change}</span>`;

            }


            // 1. 扣除對應庫存

            cart.forEach(c => {

                const target = items.find(i => i.id === c.id);

                if (target && target.stock > 0) {

                    target.stock -= 1;

                }

            });

            saveItems();


            // 2. 寫入結帳紀錄

            const now = new Date();

            const timeStr = now.toTimeString().slice(0, 8);

            const dateStr = now.toISOString().slice(0, 10);

            const record = {

                orderId: String(Date.now()),

                date: dateStr,

                time: timeStr,

                items: JSON.parse(JSON.stringify(cart)),

                paymentMethod: paymentMethod,

                isFriend: isFriendActive,

                friendLabel: isFriendActive ? friendConfig.label : '',

                discount: discount,

                total: finalTotal

            };

            orderHistory.unshift(record);

            saveHistory();


            // 3. 顯示結帳完成提示

            showNotice(

                '結帳完成！',

                `付款方式：${paymentMethod}${isFriendActive ? `（${friendConfig.label}優惠）` : ''}<br>結帳總額：<b>$${finalTotal}</b>${changeHtml}`

            );


            // 4. 清空當前購物車並刷新畫面

            clearCurrentOrder();

            renderProducts();

        }

        }


        // 單筆作廢並自動回補庫存

        function voidSingleOrder(orderId) {

            const target = orderHistory.find(r => r.orderId === orderId);

            if (!target) return;


            openConfirmDialog(

                '確定作廢此筆交易？',

                `作廢後將扣除營收 $${target.total}，並自動將該筆甜點數量加回庫存。`,

                function() {

                    target.items.forEach(soldItem => {

                        const prod = items.find(i => i.id === soldItem.id);

                        if (prod) prod.stock += 1;

                    });

                    orderHistory = orderHistory.filter(r => r.orderId !== orderId);

                    saveItems();

                    saveHistory();

                    renderProducts();

                    renderHistoryReportContent();

                }

            );

        }


        // 歸零今日營收紀錄

        function promptResetTodayIncome() {

            if (orderHistory.length === 0) return;

            openConfirmDialog(

                '確定要歸零今日收入嗎？',

                '這將會清空今天所有的結帳紀錄並將今日收入歸零（不會更動目前庫存）。',

                function() {

                    orderHistory = [];

                    saveHistory();

                    renderHistoryReportContent();

                }

            );

        }


        // 匯出 CSV 報表 (支援 Excel / iPad Numbers 繁體中文不亂碼)

        function exportCSVReport() {

            if (orderHistory.length === 0) {

                showNotice('尚無資料', '目前還沒有任何結帳紀錄可以匯出喔！');

                return;

            }


            let csv = '\uFEFF單號,日期,時間,付款方式,優惠標記,折抵金額,結帳總額,購買品項明細\n';

            const chronological = [...orderHistory].reverse();


            chronological.forEach((rec, i) => {

                const counts = {};

                rec.items.forEach(it => {

                    const k = `${it.flavor}${it.category}`;

                    counts[k] = (counts[k] || 0) + 1;

                });

                const detail = Object.entries(counts).map(([k, v]) => `${k}x${v}`).join(' / ');

                const friendStr = rec.isFriend ? (rec.friendLabel || '朋友') : '一般';

                csv += `${i + 1},${rec.date},${rec.time},${rec.paymentMethod},${friendStr},${rec.discount || 0},${rec.total},"${detail}"\n`;

            });


            const totalSum = orderHistory.reduce((s, r) => s + r.total, 0);

            const cashSum = orderHistory.filter(r => r.paymentMethod === '現金').reduce((s, r) => s + r.total, 0);

            const transferSum = orderHistory.filter(r => r.paymentMethod === '轉帳').reduce((s, r) => s + r.total, 0);

            csv += `\n總計,,,現金:$${cashSum},轉帳:$${transferSum},總營收,$${totalSum},共 ${orderHistory.length} 筆\n`;


            const blob = new Blob([csv], { type: 'text/csv;charset=utf-8;' });

            const url = URL.createObjectURL(blob);

            const a = document.createElement('a');

            const today = new Date().toISOString().slice(0, 10);

            a.href = url;

            a.download = `被狗咬一口_今日報表_${today}.csv`;

            document.body.appendChild(a);

            a.click();

            document.body.removeChild(a);

            URL.revokeObjectURL(url);

        }


        // 複製文字報表（方便在 iPad 直接貼到 LINE 或備忘錄）

        function copyTextReport() {

            const today = new Date().toISOString().slice(0, 10);

            const totalSum = orderHistory.reduce((s, r) => s + r.total, 0);

            const cashSum = orderHistory.filter(r => r.paymentMethod === '現金').reduce((s, r) => s + r.total, 0);

            const transferSum = orderHistory.filter(r => r.paymentMethod === '轉帳').reduce((s, r) => s + r.total, 0);


            // 統計各商品今日售出總數

            const soldSummary = {};

            orderHistory.forEach(r => {

                r.items.forEach(it => {

                    const name = `${it.flavor}${it.category}`;

                    soldSummary[name] = (soldSummary[name] || 0) + 1;

                });

            });

            const soldLines = Object.entries(soldSummary).map(([k, v]) => `・${k}：售出 ${v} 個`).join('\n') || '・尚無售出紀錄';

            const stockLines = items.map(it => `・${it.flavor}${it.category}：剩餘 ${it.stock} 個`).join('\n');


            const text = `🐶【被狗咬一口】市集結帳報表 (${today})\n` +

                         `━━━━━━━━━━━━━━\n` +

                         `💰 今日總收入：$${totalSum}\n` +

                         `💵 現金收入：$${cashSum}\n` +

                         `📱 轉帳收入：$${transferSum}\n` +

                         `🧾 總交易筆數：${orderHistory.length} 筆\n` +

                         `━━━━━━━━━━━━━━\n` +

                         `📦【今日銷售統計】\n${soldLines}\n` +

                         `━━━━━━━━━━━━━━\n` +

                         `🧁【目前剩餘庫存】\n${stockLines}`;


            const ta = document.createElement('textarea');

            ta.value = text;

            document.body.appendChild(ta);

            ta.select();

            document.execCommand('copy');

            document.body.removeChild(ta);


            showNotice('已複製文字報表！', '報表已複製到剪貼簿，現在可以直接打開 LINE 或備忘錄貼上囉！');

        }
